git commands
--------------------------------------------

# 1. Deshacemos el commit fallido pero mantenemos tus archivos intactos
git reset --soft HEAD~1

# 2. Agregamos los archivos corregidos al área de preparación
git add pregunta_ia.py lista_modelos.py

# 3. Creamos un nuevo commit limpio (sin secretos adentro)
git commit -m "feat: implement environment variables for secure API key management"

# 4. Empujamos el código de forma segura a GitHub
git push -u origin main
